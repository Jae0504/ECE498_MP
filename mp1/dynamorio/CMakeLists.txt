cmake_minimum_required(VERSION 3.18)
project(mp1_dynamorio_analysis LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
find_package(DynamoRIO REQUIRED)
# Required once at directory scope when linking multiple static clients.
# Initialize upstream flag bookkeeping, preserving flags for these application targets.
set(mp1_flag_vars CMAKE_C_FLAGS CMAKE_CXX_FLAGS CMAKE_C_FLAGS_RELEASE CMAKE_CXX_FLAGS_RELEASE)
foreach(flag IN LISTS mp1_flag_vars)
  set(mp1_saved_${flag} "${${flag}}")
endforeach()
configure_DynamoRIO_global(ON ON)
foreach(flag IN LISTS mp1_flag_vars)
  set(${flag} "${mp1_saved_${flag}}")
endforeach()
# This source build exports RELEASE, but upstream maps consumer RELEASE to
# RELWITHDEBINFO. Fix the mapping in our project, leaving upstream untouched.
get_property(dr_imports DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(dr_target IN LISTS dr_imports)
  get_target_property(dr_release ${dr_target} IMPORTED_LOCATION_RELEASE)
  if(dr_release)
    set_property(TARGET ${dr_target} PROPERTY MAP_IMPORTED_CONFIG_RELEASE Release)
  endif()
endforeach()
set(MP1_NATIVE_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../build-dynamorio-native" CACHE PATH "Existing native MP1 build")
add_library(mp1_kernels STATIC IMPORTED)
set_target_properties(mp1_kernels PROPERTIES IMPORTED_LOCATION "${MP1_NATIVE_BUILD}/libcpu_kernels.a")
add_executable(roi_driver roi_driver.cpp ../cpu/tests.cpp)
target_include_directories(roi_driver PRIVATE ../include)
target_compile_options(roi_driver PRIVATE -O3 -march=native -ffp-contract=fast)
target_link_libraries(roi_driver mp1_kernels OpenMP::OpenMP_CXX)
configure_DynamoRIO_static(roi_driver)
use_DynamoRIO_static_client(roi_driver drmemtrace_static)
add_executable(trace_metrics trace_metrics.cpp)
use_DynamoRIO_drmemtrace(trace_metrics)
target_link_libraries(trace_metrics drmemtrace_analyzer)
add_executable(fma_ceiling fma_ceiling.cpp)
target_include_directories(fma_ceiling PRIVATE ../include)
target_compile_options(fma_ceiling PRIVATE -O3 -march=native)
target_link_libraries(fma_ceiling OpenMP::OpenMP_CXX)

add_executable(attention_phase_driver attention_phase_driver.cpp ../cpu/tests.cpp)
target_include_directories(attention_phase_driver PRIVATE ../include)
target_compile_options(attention_phase_driver PRIVATE -O3 -march=native -ffp-contract=fast)
target_link_libraries(attention_phase_driver mp1_kernels OpenMP::OpenMP_CXX)
configure_DynamoRIO_static(attention_phase_driver)
use_DynamoRIO_static_client(attention_phase_driver drmemtrace_static)
